太牛了。

先将边权放到矿工和黄金上。

费用流的建模很显然，树上父亲向儿子连流量 $inf$，费用 $0$ 的边，源点向矿工连流量为 $1$，费用为 $v$ 的边，黄金向汇点连流量为 $1$，费用为 $v$ 的边。

考虑如何模拟这个费用流。新加一个矿工后，它可能会去匹配一个未匹配过的黄金，也可能去抢一个矿工的黄金。那么抢矿工的黄金可以看成将原来的矿工变成一个黄金，权值为负。黄金也同理可以看成矿工，并去匹配一个最大的矿工。注意不可能增广一次后还能继续增广，易证。

所以，我们现在要维护的是新加矿工/黄金后所有能到的黄金/矿工的权值最大值。

先考虑加一个矿工。根据建模，它可以随意地向儿子走，并且可以走一些有流量的反向边。那么他能走到的点就是往到根路径上第一个反向边流量 $=0$ 的边对应的儿子，然后这个点的子树都可以随意走到。反向边流量只会区间加/减，所以可以线段树二分找出第一个流量为 $0$ 的点，再维护 dfs 序区间下的最大权值黄金。之后这个矿工变成了黄金，直接单点修改即可。我们可以通过 set 求出在同一个点上的黄金权值最大值。

现在考虑加一个黄金。根据建模，它可以随意匹配向上的矿工，也可以向下走一些反向边流量为正的边（相当于矿工走反向边）。考虑怎么维护这个玩意。我们考虑树剖，再在每个点维护所有轻子树中能到达该点（路径上反向边都有流量）的矿工权值最大值（设为 $w_i$）。这个东西如何更新？在一条重链中求出 dfn 从 top 连续反向边有流量的长度，求出这个 dfs 序区间的 $w$ 的最大值，再贡献到这条重链顶端的父亲即可。答案即为这个黄金到根路径的 $w$ 的最大值。发现 $w$ 同样需要开 set 维护。

对于黄金变成的矿工，我们可以先撤销这个点 $x$ 对应的所有矿工对 $x$ 到根的路径的贡献并在线段树上也赋值为负无穷，加入这个点后再重新计算更新。

在代码实现中有更加优的写法。

$O(n\log^2 n)$。

一个疑问：为什么判一哈增广的贡献 $\le 0$ 就不匹配会错啊/qd。

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int inf = 1e9;
const ll INF = 1e15;
const int N = 1e5;
inline int read() {
	int s = 0,f = 1;char ch = getchar();
	while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
	while (isdigit(ch)) s = (s << 3) + (s << 1) + ch - '0', ch = getchar();
	return s*f;
}
int n,Q,head[N + 10],cnt,vis[N + 10];
struct edge {
	int v,w,nxt;
}ed[N * 2 + 10];
ll dis[N + 10],ans;
int dep[N + 10],fa[N + 10],sz[N + 10],son[N + 10],top[N + 10],down[N + 10],id[N + 10],rnk[N + 10],tot;
void add(int u,int v,int w) {
	ed[++cnt] = {v,w,head[u]};
	head[u] = cnt;
}
void dfs(int x,int f) {
	fa[x] = f, sz[x] = 1;
	for (int i = head[x];i;i = ed[i].nxt) {
		int v = ed[i].v;
		dep[v] = dep[x] + 1;
		dis[v] = dis[x] + ed[i].w;
		dfs(v,x);
		sz[x] += sz[v];
		if (sz[son[x]] < sz[v]) son[x] = v;
	}
}
void dfs2(int x,int tp) {
	id[x] = ++tot, rnk[tot] = x, top[x] = tp;
	if (!son[x]) {
		down[x] = id[x];
		return;
	}
	dfs2(son[x],tp);
	down[x] = down[son[x]];
	for (int i = head[x];i;i = ed[i].nxt) {
		int v = ed[i].v;
		if (v != son[x]) dfs2(v,v);
	}
}
struct node {
	ll v;
	int x;
	inline bool operator < (const node &a) const {
		return v < a.v;
	}
};
struct Seg {
	int l,r,mn,tag;
	node mx[2];
}t[N * 4 + 10];
void build(int x,int l,int r) {
	t[x].l = l, t[x].r = r, t[x].mx[0] = t[x].mx[1] = {-INF,0};
	if (l == r) return;
	int mid = l + r >> 1;
	build(x<<1,l,mid), build(x<<1|1,mid + 1,r);
}
void f(int x,int s) {
	t[x].mn += s, t[x].tag += s;
}
void pushdown(int x) {
	f(x<<1,t[x].tag), f(x<<1|1,t[x].tag);
	t[x].tag = 0;
}
void modify(int x,int k,node s,int fl) {
	if (t[x].l == t[x].r) {
		t[x].mx[fl] = s;
		return;
	}
	int mid = t[x].l + t[x].r >> 1;
	if (mid >= k) modify(x<<1,k,s,fl);
	else modify(x<<1|1,k,s,fl);
	t[x].mx[0] = max(t[x<<1].mx[0],t[x<<1|1].mx[0]);
	t[x].mx[1] = max(t[x<<1].mx[1],t[x<<1|1].mx[1]);
}
void modifyf(int x,int l,int r,int s) {
	if (l > r) return;
	if (t[x].l >= l && r >= t[x].r) {
		f(x,s);
		return;
	}
	pushdown(x);
	int mid = t[x].l + t[x].r >> 1;
	if (mid >= l) modifyf(x<<1,l,r,s);
	if (mid < r) modifyf(x<<1|1,l,r,s);
	t[x].mn = min(t[x<<1].mn,t[x<<1|1].mn);
}
node query(int x,int l,int r,int fl) {
	if (l > r) return {-INF,0};
	if (t[x].l >= l && r >= t[x].r) return t[x].mx[fl];
	int mid = t[x].l + t[x].r >> 1;
	node res = {-INF,0};
	if (mid >= l) res = query(x<<1,l,r,fl);
	if (mid < r) res = max(res,query(x<<1|1,l,r,fl));
	return res;
}
int queryr(int x,int l,int r) {// 区间从右往左第一个 <0 的位置 因为前面先减了1 
	if (l > r || t[x].mn >= 0) return -1;
	if (t[x].l == t[x].r) return t[x].l;
	int mid = t[x].l + t[x].r >> 1;
	pushdown(x);
	if (mid < r) {
		int t = queryr(x<<1|1,l,r);
		if (t != -1) return t;
	}
	return mid >= l ? queryr(x<<1,l,r) : -1;
}
int queryl(int x,int l,int r) {// 区间从左往右第一个 <=0 的位置 
	if (l > r) return -1;
	if (t[x].l == t[x].r) return t[x].mn > 0 ? -1 : t[x].l;
	int mid = t[x].l + t[x].r >> 1;
	pushdown(x);
	if (mid >= l && t[x<<1].mn <= 0) {
		int t = queryl(x<<1,l,r);
		if (t != -1) return t;
	}
	return mid < r ? queryl(x<<1|1,l,r) : -1;
}
int get1(int x) {//find a gold 
	while (x) {
		int t = queryr(1,id[top[x]],id[x]);
		if (t != -1) return rnk[t];
		x = fa[top[x]];
	}
	return 1;
}
multiset<ll> S[2][N + 10];//0 是矿工 1 是黄金
multiset<node> W[N + 10];//矿工轻儿子上来的最大值 
node get0(int x) {//重链前缀的最大值 find a miner 
	int tt = queryl(1,id[x],down[x]);
	return query(1,id[x],tt == -1 ? down[x] : tt - 1,0);
}
void updf(int x,int v) {
	while (x) {
		modifyf(1,id[top[x]],id[x],v);
		x = fa[top[x]];
	}
}
node gtop(int x) {
	return W[x].size() ? *W[x].rbegin() : (node){-INF,0};
}
ll gtop(int t,int x) {
	return S[t][x].size() ? *S[t][x].rbegin() : -INF;
}
void delv(int x) {
	while (x) {
		node v = get0(top[x]);
		if (fa[top[x]]) W[fa[top[x]]].erase(W[fa[top[x]]].find(v));
		modify(1,id[x],{-INF,0},0);
		x = fa[top[x]];
	}
}
void addv(int x) {
	while (x) {
		modify(1,id[x],gtop(x),0);
		if (fa[top[x]]) W[fa[top[x]]].insert(get0(top[x]));
		x = fa[top[x]];
	}
}
void add0(int u,ll v) {
	delv(u);
	S[0][u].insert(v);
	W[u].insert({v,u});
	addv(u);
}
void add1(int u,ll v) {
	S[1][u].insert(v);
	modify(1,id[u],{gtop(1,u),u},1);
}
//how will the flow changes?
//when a miner goes up, flow sub 1
//when he goes down to mine a gold, flow add 1
//so we only need to make the flow from miner to root sub 1, and make the flow from gold to root add 1
void update0(int x) {//a miner is matched!
	//we dont care about the one who matched is the new one
	//we only need to match the greatest one
	//turn it to gold
	//del a miner
	//the flow from x to root sub 1
	delv(x);
	ll y = gtop(0,x);
	ans += y;
	S[0][x].erase(S[0][x].find(y));
	W[x].erase(W[x].find({y,x}));
	add1(x,-y);
	updf(x,-1);
	addv(x);
}
void update1(int x) {//the same
	ll y = gtop(1,x);
	ans += y;
	S[1][x].erase(S[1][x].find(y));
	modify(1,id[x],{gtop(1,x),x},1);
	add0(x,-y);
	delv(x);
	updf(x,1);
	addv(x);
}
signed main() {
	n = read(), Q = read();
	for (int i = 1;i < n;i ++ ) {
		int u = read(),v = read(),w = read();
		add(u,v,w);
		vis[v] = 1;
	}
	dfs(1,0), dfs2(1,1);
	for (int i = 1;i <= n;i ++ ) {
		if (top[i] == i) W[fa[i]].insert({-INF,0});
	}
	build(1,1,n);
	ans = 0;
	while (Q -- ) {
		int op = read(),u = read();
		ll v = read() + (op == 1 ? -1 : 1) * dis[u];
		if (op == 1) {
			//add a miner u
			add0(u,v);
			if (gtop(0,u) != v) goto bre;
			update0(u);
			//u and y matched successfully
			//turn u to gold and turn y to miner
			//add and del some edge's flow
			//del the cont from u/y, add/del flow, add the cont from u/y.
			//you dont need to promise that cont > 0
			//because if you let the miner be a gold at first
			//the gold at least can match the miner and the cont is 0.
			int x = get1(u);
			node xv = query(1,id[x],id[x] + sz[x] - 1,1);
			update1(xv.x);
		}
		if (op == 2) {
			add1(u,v);
			if (gtop(1,u) != v) goto bre;
			update1(u);
			update0(get0(1).x);
		}
		bre:;
		printf("%lld\n",ans);
	}
	return 0;
}
```
