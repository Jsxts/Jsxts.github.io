我有一个朋友码力太差，在赛后 2h 调出来后直接破防。

考虑第一个数怎么找。首先发现答案并不满足单调性，所以只能从大到小枚举然后 check，判断一个数在 $x$ 子树中花费 $K$ 的代价能否被钦定第一个选。

这题的关键在于如何 check。我们考虑从这个点往上跳 fa，每次跳到一个位置分讨：

- 若 fa 的右子树为当前节点，则对于 fa 的左子树来说，在 fa 上打标记是无效的，只需要判断在左子树内能否找到一个点满足：只考虑左子树内，选出一个 $>v$ 的数作为第一个数的最小花费。将其设为 $dp_{x,v}$，发现总状态数为 $O(2^nn)$，所以可以直接暴力转移得出。

- 若 fa 的左子树为当前节点，则有多一种选择：直接标记 fa。在这两者中取较小值即可。

得到第一个数后，我们考虑用类似的思路解决剩下的问题。首先肯定是从下往上每个子树依次处理，然后同样的跳 fa，似乎直接递归进入其他子树贪心的选就好了？？

不是的，由于这样可能会使到最后反而满足不了最初始的对整个子树的第一个数 $p$ 的限制。不过幸运的是，这里仍然能贪心：我们只需要保留使得后面能使 $p$ 满足条件的最小代价，其他可以全部往另一个子树里扔（越靠前优先级越高），所以预处理每个 fa 到根所需最小值的后缀和即可。

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 17;
const int M = 1 << N;
typedef long long ll;
const int inf = 1e9;
const ll INF = 1e18;
ll read() {
	ll s = 0,f = 1;char ch = getchar();
	while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
	while (isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
	return s * f;
}
int n,m,q[1 << N],ans[1 << N],tt;
ll K,w[1 << N];
int tot;
vector<pair<int,int> > vc[M + 10];
void init(int x) {
	if (x >= m) vc[x].push_back({q[x],x});
	else {
		init(x<<1), init(x<<1|1);
		for (auto v : vc[x << 1]) vc[x].push_back(v);
		for (auto v : vc[x << 1 | 1]) vc[x].push_back(v);
		sort(vc[x].begin(),vc[x].end());
	}
}
ll check(int rt,int x,ll K);
vector<ll> vc2[M + 10];
vector<pair<int,ll> > dp[M + 10],dp2[M + 10];
ll get2(int rt,int v) {
	if (rt >= m) return q[rt] > v ? 0 : -1;
	int x = upper_bound(dp2[rt].begin(),dp2[rt].end(),make_pair(v,0ll)) - dp2[rt].begin();
	return dp2[rt][x].second;
}
ll suf[M + 10];
void dfs(int rt) {
	if (rt >= m) {
		dp[rt].push_back({q[rt],0});
		return;
	}
	dfs(rt << 1), dfs(rt << 1 | 1);
	suf[dp[rt << 1 | 1].size()] = INF;
	for (int j = dp[rt << 1 | 1].size() - 1;~j;j -- )
		suf[j] = min(suf[j + 1],dp[rt << 1 | 1][j].second);
	int r = 0;
	for (auto v : dp[rt << 1]) {
		while (r < dp[rt << 1 | 1].size() && dp[rt << 1 | 1][r].first < v.first) r ++;
		dp[rt].push_back({v.first,v.second + min(w[rt],suf[r])});
	}
	suf[dp[rt << 1].size()] = INF;
	for (int j = dp[rt << 1].size() - 1;~j;j -- )
		suf[j] = min(suf[j + 1],dp[rt << 1][j].second);
	r = 0;
	for (auto v : dp[rt << 1 | 1]) {
		while (r < dp[rt << 1].size() && dp[rt << 1][r].first < v.first) r ++;
		dp[rt].push_back({v.first,v.second + suf[r]});
	}
	sort(dp[rt].begin(),dp[rt].end());
	dp2[rt].resize(dp[rt].size());
	dp2[rt].push_back({inf,(ll)1e12});
	for (int i = dp2[rt].size() - 2;~i;i -- ) dp2[rt][i].first = dp[rt][i].first, dp2[rt][i].second = min(dp2[rt][i + 1].second,dp[rt][i].second);
}
ll tmp2[M + 10];
ll check(int rt,int x,ll K) {
	vector<ll> vc;
	int tx = x;
	while (x != rt) {
		int fa = x / 2;
		if (fa * 2 + 1 == x) {
			ll p = get2(fa * 2,q[tx]);
			if (p == -1) return 0;
			K -= p;
			vc.push_back(K);
			if (K < 0) return 0;
		}
		else {
			ll p = get2(fa * 2 + 1,q[tx]);
			if (p == -1 || p > w[fa]) K -= w[fa], vc.push_back(K + w[fa]);
			else K -= p, vc.push_back(K);
			if (K < 0) return 0;
		}
		x = fa;
	}
	for (int i = tx,j = 0;i != rt;i >>= 1, j ++ ) tmp2[i] = vc[j] - K;
	return K + 1;
}
map<pair<int,ll>,ll> mp;
ll solve(int x,ll K) {
	int p;
	for (int i = vc[x].size() - 1;~i;i -- ) {
		if (check(x,vc[x][i].second,K)) {
			p = vc[x][i].second;
			break;
		}
	}
	vector<ll> vc;
	for (int i = p;i != x;i >>= 1) vc.push_back(tmp2[i]);
	ans[++tt] = q[p];
	int tp = p,now = 0;
	while (p != x) {
		int fa = p / 2;
		if (fa * 2 + 1 == p) {
			K = solve(fa * 2,K - vc[now]) + vc[now];
		}
		else {
			ll p = get2(fa * 2 + 1,q[tp]);
			if (p != -1 && p <= w[fa])
				K = solve(fa * 2 + 1,K - vc[now]) + vc[now];
			else {
				if (p != -1 && K - vc[now] + w[fa] >= p) K = solve(fa * 2 + 1,K - vc[now] + w[fa]) + vc[now] - w[fa];
				else K = solve(fa * 2 + 1,K - vc[now]) + vc[now] - w[fa];
			}
		}
		now ++;
		p = fa;
	}
	return K;
}
int main() {
	int T = read();
	while (T -- ) {
		n = read(), K = read(), m = 1 << n;
		for (int i = 1;i < m;i ++ ) w[i] = read();
		for (int i = 1;i <= m;i ++ ) q[i + m - 1] = read();
		for (int i = 1;i <= 2 * m;i ++ ) vc[i].clear(), dp[i].clear(), dp2[i].clear();
		init(1);
		dfs(1);
		tt = 0;
		solve(1,K);
		for (int i = 1;i <= m;i ++ ) {
			printf("%d",ans[i]);
			if (i != m) putchar(' ');
		}
		puts("");
	}
	return 0;
}
```
