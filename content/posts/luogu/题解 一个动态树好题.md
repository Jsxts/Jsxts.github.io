XSY 1662

考虑将 $x\rightarrow p_x$ 连边，连完后形成了基环树森林。对于环上的每个点（设点数为 $n$），手动模拟就会发现一直代入最后会有个形如

$$x_i=x_i\prod_{j=1}^n k_j+\sum_{j=1}^nb_j\prod_{l=1}^{j-1}k_l$$

的方程，这个方程可以用 exgcd 或者逆元求解。注意 $k=1$ 时要判无解或多解。

之后，只要求出环上任意一个点，不在环上的都可以通过不断代入的方法求出解，形式与上面类似。

那么如何维护上面的两个系数呢？容易发现这是可合并的，用线段树之类的就可以维护，但是因为这棵基环树形态会变，只能考虑动态树（或者动态仙人掌，但是我不会）。动态树维护的是一棵树，所以要在环上断掉一条边，并且把断掉的边记录下来，然后把一个端点设为根，查询时就先求出根的解，之后再求出查询点到根的链的最终系数，查询点的解就出来了。

在修改的时候，要分修改的点是根/在环上/其他来讨论，具体手动模拟。

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int inf = 2e9;
int read() {
	int s = 0,f = 1;char ch = getchar();
	while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
	while (isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
	return s*f;
}
void write(int x) {
	if (x < 0) putchar('-'), x = -x;
	if (x > 9) write(x / 10);
	putchar(x % 10 + 48);
}
const int mod = 10007;
int exgcd(int a,int b,int &x,int &y) {
	if (!b) {
		x = 1, y = 0;
		return a;
	}
	int gc = exgcd(b,a % b,x,y);
	int t = x;
	x = y;
	y = t - a / b * y;
	return gc;
}
int qpow(int a,int b) {
	int res = 1;
	while (b) {
		if (b & 1) res = 1ll * res * a % mod;
		b >>= 1, a = 1ll * a * a % mod;
	}
	return res;
}
int fa[100010];
struct Link_Cut_Tree {
	struct nd {
		int k,b;
		friend nd operator + (nd a,nd b) {
			return {a.k * b.k % mod,(b.k * a.b + b.b) % mod};
		}
	};
	struct node {
		int ch[2],fa,sfa,tag;
		nd v,s;
	}t[100010];
	#define ls t[x].ch[0]
	#define rs t[x].ch[1]
	bool c(int x) {
		return t[t[x].fa].ch[1] == x;
	}
	bool isroot(int x) {
		return t[t[x].fa].ch[0] != x && t[t[x].fa].ch[1] != x;
	}
	void pushup(int x) {
		t[x].s = t[x].v;
		if (ls) t[x].s = t[ls].s + t[x].s;
		if (rs) t[x].s = t[x].s + t[rs].s;
	}
	void rev(int x) {
		swap(ls,rs), t[x].tag ^= 1;
	}
	void pushdown(int x) {
		if (t[x].tag) {
			if (ls) rev(ls);
			if (rs) rev(rs);
			t[x].tag = 0;
		}
	}
	void rotate(int x) {
		int y = t[x].fa,z = t[y].fa,k = c(x);
		if (!isroot(y)) t[z].ch[c(y)] = x;
		t[x].fa = z;
		t[y].ch[k] = t[x].ch[!k], t[t[x].ch[!k]].fa = y;
		t[x].ch[!k] = y, t[y].fa = x;
		pushup(y), pushup(x);
	}
	void pushall(int x) {
		if (!isroot(x)) pushall(t[x].fa);
		pushdown(x);
	}
	void splay(int x) {
		pushall(x);
		for (int f;f = t[x].fa, !isroot(x);rotate(x))
			if (!isroot(f)) rotate(c(f) == c(x) ? f : x);
	}
	void access(int x) {
		for (int i = 0;x;x = t[i = x].fa)
			splay(x), rs = i, pushup(x);
	}
	void makeroot(int x) {
		access(x), splay(x), rev(x);
	}
	void split(int x,int y) {
		makeroot(x), access(y), splay(y);
	}
	int findroot(int x) {
		access(x), splay(x), pushdown(x);
		while (ls) x = ls, pushdown(x);
		splay(x);
		return x;
	}
	void link(int x,int y) {
		makeroot(x);
		t[x].fa = y;
	}
	void cut(int x) {
		access(x), splay(x);
		t[ls].fa = 0;
		ls = 0;
	}
	int query(int x) {
		int rt = findroot(x),y = t[rt].sfa;
//		cout << rt << endl;
		split(rt,y);
		int k = t[y].s.k,b = t[y].s.b,x1;
		if (k == 1) {
			if (b == 0) return -2;
			else return -1;
		}
		else if (k) {
			x1 = (b + mod) % mod * qpow((1 - k + mod) % mod,mod - 2) % mod;
		}
		else x1 = b;
		split(rt,x);
		return (t[x].s.k * x1 + t[x].s.b) % mod;
	}
	bool pd(int rt,int x) {
		int y = t[rt].sfa;
		if (x == y) return 1;
		access(y), splay(y);
		splay(x);
		return !isroot(y);
	}
	void modify(int x,int k,int b,int p) {
		t[x].v = {k,b};
		int rt = findroot(x);
		if (x == rt) {
			if (findroot(p) == x) t[x].sfa = p;
			else t[x].sfa = 0, link(x,p);
		}
		else if (pd(rt,x)) {
			cut(x);
			link(rt,t[rt].sfa);
			t[rt].sfa = 0;
			if (findroot(p) == x) t[x].sfa = p;
			else link(x,p);
		}
		else {
			cut(x);
			if (findroot(p) == x) t[x].sfa = p;
			else link(x,p);
		}
	}
}T;
int find(int x) {
	return fa[x] == x ? x : fa[x] = find(fa[x]);
}
int main() {
	int n = read();
	for (int i = 1;i <= n;i ++ ) fa[i] = i;
	for (int i = 1;i <= n;i ++ ) {
		int k = read(),p = read(),b = read();
		T.t[i].v = {k,b};
		if (find(p) == find(i)) T.t[i].sfa = p;
		else {
			T.t[i].fa = p;
			fa[find(i)] = find(p);
		}
	}
	int q = read();
	while (q -- ) {
		char op[2];
		scanf("%s",op);
		if (op[0] == 'A') {
			int x = read();
			printf("%d\n",T.query(x));
		}
		if (op[0] == 'C') {
			int x = read(),k = read(),p = read(),b = read();
			T.modify(x,k,b,p);
		}
	}
	return 0;
}
```
