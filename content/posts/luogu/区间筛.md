原理和埃氏筛相同，即找到一个素数 $p$ 后将区间中所有 $p$ 的倍数划去。

将 $\sqrt r$ 内的素数预处理出来，然后用它们去更新 $[l,r]$ 中每个数是否为素数。

$O(k\log\log r + \sqrt r)$。其中 $k$ 为区间长度。

```cpp
int visp[N + 10],vis[N + 10];
void sieve(int l,int r) {
	for (ll i = 2;i * i <= r;i ++ ) {
		if (!visp[i]) {
			for (ll j = i + i;j * j <= r;j += i) visp[j] = 1;
			for (ll j = i * max(2,(int)ceil(l * 1.0 / i));j <= r;j += i) vis[j - l + 1] = 1;
		}
	}
	if (l == 1) vis[1] = 1;
}
```

区间筛积性函数：

此时因为涉及到每个质因子的次数，要枚举每个质因子，然后筛它的每个次方。可以先线性筛预处理质数。

这里次方从大到小枚举据说可以减小常数。

例：区间筛 $i^k$：

```cpp
int vis[N + 10],g[N + 10],tmp[N + 10];
int p[N + 10],cnt,pk[N + 10],inv[N + 10];
bool visp[N + 10];
void init() {
	pk[1] = 1;
	for (int i = 2;i <= N;i ++ ) {
		if (!visp[i]) p[++cnt] = i, pk[i] = qpow(i,k);
		for (int j = 1;j <= cnt && p[j] * i <= N;j ++ ) {
			visp[i * p[j]] = 1;
			pk[i * p[j]] = 1ll * pk[i] * pk[p[j]] % mod;
			if (i % p[j] == 0) break;
		}
	}
	inv[0] = inv[1] = 1;
	for (int i = 2;i <= N;i ++ ) inv[i] = 1ll * (mod - mod / i) * inv[mod % i] % mod;
}
void sieve(int l,int r) {
	for (int i = l;i <= r;i ++ ) g[i - l] = 1, tmp[i - l] = 1;
	for (int i = 1;1ll * p[i] * p[i] <= r;i ++ ) {
		ll mul = 1,j = 1,ipk = qpow(pk[p[i]],mod - 2);
		while (r / j * j >= l) j *= p[i], mul = 1ll * mul * pk[p[i]] % mod;
		j /= p[i], mul = 1ll * mul * ipk % mod;
		for (;j > 1;j /= p[i], mul = 1ll * mul * ipk % mod)
			for (ll t = r / j * j;t >= l;t -= j) 
				if (vis[t - l] != i) vis[t - l] = i, g[t - l] = 1ll * g[t - l] * mul % mod, tmp[t - l] *= j;
	}
	for (int i = l;i <= r;i ++ ) {
		if (tmp[i - l] < i) {
			int x = i / tmp[i - l];
			g[i - l] = 1ll * g[i - l] * ((x <= N) ? pk[x] : qpow(x,k)) % mod;
		}
	}
}
```
