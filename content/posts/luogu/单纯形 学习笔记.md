目前关于单纯形只会背板以及大概知道原理和如何使用，其他还看不懂。

单纯形用来解决线性规划问题。

线性规划的标准形：

求：

$$\max z = \sum_i c_ix_i$$

限制：

- $\sum_{j=1}^na_{i,j}x_j=b_i,\forall i\in [1,m]\cap Z^+$

- $x_j\ge 0,\forall j\in [1,n]\cap Z^+$

$n$ 个变量，$m$ 个限制。

任何网络流问题转成这个形式即可套用单纯形模板求解。$\min$ 变为 $\max$、$\le,\ge$ 变为 $=$ 可以增加变量、左右取反来转化。

复杂度理论上是指数级，但是实际期望复杂度为 $O(nm^2)$（$n,m\le 1000$ 左右也能试试）。

#### 单纯形算法简要流程：

试验一组基变量在限制中求出的解 $\to$ 改变基变量 $\to$ 直到无法改变最终求出答案。

模板（不记得改头）：

```cpp
#include<bits/stdc++.h>
#define Yukinoshita namespace
#define Yukino std
using Yukinoshita Yukino;
//#define double long long
inline int read() {
	int s = 0,f = 1;char ch = getchar();
	while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
	while (isdigit(ch)) s = (s << 3) + (s << 1) + ch - '0', ch = getchar();
	return s*f;
}
const int N = 2e2;
int ans[N * 2 + 10];
struct Simplex {
	const double eps = 1e-6;
	int n,m,id[N * 5 + 10];
	double A[N * 3 + 10][N * 2 + 10];
	void pivot(int r,int c) {//交换基变量与非基变量
	    swap(id[r + n],id[c]);
	    double t = A[r][c]; 
	    A[r][c] = 1;
	    for (int i = 0;i <= n;i ++ )
	        A[r][i] /= t;
	    for (int i = 0;i <= m;i ++ )
	        if (abs(A[i][c]) > eps && i != r) {
	            t = A[i][c], A[i][c] = 0;
	            if (!t) continue;
	            for (int j = 0;j <= n;j ++ ) A[i][j] -= t * A[r][j];
	        }
	}
	bool init() {
		for (int i = 1;i <= n;i ++ ) id[i] = i;
	    while (1) {
	        int r = 0,c = 0;
	        for (int i = 1;i <= m;i ++ )
	            if (A[i][0] < -eps && (!r || rand() & 1)) r = i;
	        if (!r) return 1;
	        for (int i = 1;i <= n;i ++ )
	            if (A[r][i] < -eps && (!c || rand() & 1)) c = i;
	        if (!c) {
	            puts("Infeasible");
	            return 0;
	        }
	        pivot(r,c);
	    }
	}
	bool simplex() {
	    while (1) {
	        int r = 0,c = 0;
	        double mx = eps;
	        for (int i = 1;i <= n;i ++ )
	            if (A[0][i] > mx)
	                c = i, mx = A[0][i]; 
	        if (!c) return 1;
	        double mn = 1e18;
	        for (int i = 1;i <= m;i ++ )
	            if (A[i][c] > eps && A[i][0] / A[i][c] < mn)
	                r = i, mn = A[i][0] / A[i][c];
	        if (!r) {
	            puts("Unbounded");
	            return 0;
	        }
	        pivot(r,c);
	    }
	}
	void get_project() {
		for (int i = 1;i <= m;i ++ ) ans[id[i + n]] = floor(A[i][0] + 0.5);
	}
	//n 变量数 m 限制数 
	//A[0][i]:标准形中的 c[i]
	//A[i][j]:标准形中的 a[i][j]
	//A[i][0]:原为标准形中的 b[j]，经过算法后变为标准形中的 x[j]（即解） 
	//-A[0][0]:max z
	//注意此代码处理的限制为Σa[i][j]*x[j]<=b[i] 而非 = 
	//调用init()初始化，若所有 x[i] = 0 是一组可行解就不用初始化，之后会得到一组可行解。
```

例题：P3980 志愿者招募

将每天有多少人工作看成限制 $A_i$,将每个人工作几次看成变量 $x_j$ 将 $a_{i,j}$ 看成第 $j$ 个人是否在第 $i$ 天工作，则 $n$ 条限制为：

$$\sum_{j}a_{i,j}x_j\ge A_i$$

求 $\min z = \sum_jc_jx_j$。将 $c_j$ 取反后转为求最大。将 $a_{i,j},A_i$ 取反后限制变为 $\sum_{j}a_{i,j}x_j\le A_i$，直接套用模板即可。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
inline int read() {
	int s = 0,f = 1;char ch = getchar();
	while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
	while (isdigit(ch)) s = (s << 3) + (s << 1) + ch - '0', ch = getchar();
	return s*f;
}
const int N = 1e4;
const int eps = 0;
int n,m,id[N * 2 + 10];
ll A[N / 10 + 10][N + 10];//m*n
int ans[N];
void pivot(int r,int c) {//交换基变量与非基变量
	swap(id[r + n],id[c]);
	ll t = A[r][c]; 
	A[r][c] = 1;
	for (int i = 0;i <= n;i ++ )
		A[r][i] /= t;
	vector<int> vc;
	for (int j = 0;j <= n;j ++ )
		if (A[r][j]) vc.push_back(j);
	for (int i = 0;i <= m;i ++ )
		if (abs(A[i][c]) > eps && i != r) {
			t = A[i][c], A[i][c] = 0;
			if (!t) continue;
			for (auto j : vc) A[i][j] -= t * A[r][j];
		}
}
bool init() {
	while (1) {
		int r = 0,c = 0;
		for (int i = 1;i <= m;i ++ )
			if (A[i][0] < -eps && (!r || rand() & 1)) r = i;
		if (!r) return 1;
		for (int i = 1;i <= n;i ++ )
			if (A[r][i] < -eps && (!c || rand() & 1)) c = i;
		if (!c) {
			puts("Infeasible");
			return 0;
		}
		pivot(r,c);
	}
}
bool simplex() {
	while (1) {
		int r = 0,c = 0;
		ll mx = eps;
		for (int i = 1;i <= n;i ++ )
			if (A[0][i] > mx)
				c = i, mx = A[0][i]; 
		if (!c) return 1;
		ll mn = 1e18;
		for (int i = 1;i <= m;i ++ )
			if (A[i][c] > eps && A[i][0] / A[i][c] < mn)
				r = i, mn = A[i][0] / A[i][c];
		if (!r) {
			puts("Unbounded");
			return 0;
		}
		pivot(r,c);
	}
}
//A[0][i]:标准形中的 c[i]
//A[i][j]:标准形中的 a[i][j]
//A[j][0]:原为标准形中的 b[j]，经过算法后变为标准形中的 x[j]（即解） 
//-A[0][0]:max z
//注意此代码处理的限制为Σa[i][j]*x[j]<=b[i] 而非 = 
//调用init()初始化，若所有 x[i] = 0 是一组解就不用初始化
//调用simplex()进行算法 
int main() {
	int n = read(),m = read();
	for (int i = 1;i <= n;i ++ ) A[i][0] = -read();
	::n = m, ::m = n;
	for (int i = 1;i <= m;i ++ ) {
		int s = read(),t = read();
		A[0][i] = -read();
		for (int j = 1;j <= n;j ++ )
			A[j][i] = -(s <= j && j <= t);
	}
	init(); simplex();
	cout << A[0][0];
	return 0;
}
```
