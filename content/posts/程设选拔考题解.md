## A 矩形上下边距离最短的点对（加强版）

题意：在二维平面上有两行点：$(a_i,0)$ 和 $(b_j,1)$，满足横坐标为整数且严格递增。

在每行各取一个点，求最短距离。



题解：即求 $|a_i-b_j|$ 的最小值。对每个 $a_i$ 在 $b$ 中用 $\text{lower\_bound}$ 找到前驱和后继并更新答案即可。也可以直接扫一遍做到 $O(n)$。

## B 小明的周末

题意：有 $n$ 个数 $a_1\dots a_n$，求两个不相交且长度均为 $k$ 的区间使得区间中数的和最小，并输出两个区间的位置。



题解：预处理出 $i\sim n$ 中长度为 $k$ 的区间的最小和 $s_i$，然后枚举 $i$，用以 $i$ 结尾，长度为 $k$ 的区间和加上 $s_{i+1}$ 更新答案。要求输出位置就用 $\text{pair}$ 存区间位置。

## C 网格游戏

![](C:\Users\yangsonghan\my-blog\static\images\程设选拔考题解\T3.png)

$n,m\leq 400,k\leq 20$。

题解：

采用动态规划，并且记录当前是否正在使用技能以及走路的方向。每次移动的时候考虑这一步是否发动技能。

设 $dp_{i,j,k,0/1,0/1}$ 表示当前位于 $(i,j)$，还剩 $k$ 次技能可使用，上一步从左边/上面走过来，当前不处于/处于发动技能期间。

转移没啥好说，根据定义即可。注意如果发动技能，由于技能不是在移动时发动而是在格点发动，所以我们认为在上一个点就已经发动，要加上上一个点的得分。在 $(1,1)$ 也能发动，于是初始化要注意。

直接交会 MLE，于是滚动优化即可。

```c++
#include<bits/stdc++.h>
using namespace std;
const int N = 400;
const int K = 200;
typedef long long ll;
const int inf = 1e9;
int read() {
    int s = 0,f = 1;char ch = getchar();
    while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
    while (isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
    return s * f;
}
int n,m,kk,a[N + 10][N + 10];
int dp[2][N + 10][K + 10][2][2];
int main() {
    n = read(), m = read(), kk = read();
    for (int i = 1;i <= n;i ++ )
        for (int j = 1;j <= m;j ++ ) a[i][j] = read();
    for (int i = 0;i <= 1;i ++ )
        for (int j = 0;j <= m + 1;j ++ )
            for (int k = 0;k <= kk;k ++ ) dp[i][j][k][0][0] = dp[i][j][k][0][1] = dp[i][j][k][1][0] = dp[i][j][k][1][1] = -inf;
    dp[1][1][kk][0][0] = dp[1][1][kk][1][0] = a[1][1];
    dp[1][1][kk-1][0][1] = dp[1][1][kk-1][1][1] = a[1][1]*2;
    int i=1,now=1;
    for (i = 1,now = 1;now <= n;now ++ ,i ^= 1)
        for (int j = 1;j <= m;j ++ ) {
            for (int k = 0;k <= kk;k ++ ) {
                if (j > 1) {
                    dp[i][j][k][0][0] = max(dp[i][j][k][0][0],dp[i][j - 1][k][1][1] + a[now][j]);
                    dp[i][j][k][0][0] = max(dp[i][j][k][0][0],dp[i][j - 1][k][0][1] + a[now][j]);
                    dp[i][j][k][0][0] = max(dp[i][j][k][0][0],dp[i][j - 1][k][1][0] + a[now][j]);
                    dp[i][j][k][0][0] = max(dp[i][j][k][0][0],dp[i][j - 1][k][0][0] + a[now][j]);
                    if (k < kk) {
                        dp[i][j][k][0][1] = max(dp[i][j][k][0][1],dp[i][j - 1][k + 1][0][0] + a[now][j] * 2 + a[now][j-1]);
                        dp[i][j][k][0][1] = max(dp[i][j][k][0][1],dp[i][j - 1][k + 1][1][0] + a[now][j] * 2 + a[now][j-1]);
                        dp[i][j][k][0][1] = max(dp[i][j][k][0][1],dp[i][j - 1][k + 1][1][1] + a[now][j] * 2);
                        
                    }
                    dp[i][j][k][0][1] = max(dp[i][j][k][0][1],dp[i][j - 1][k][0][1] + a[now][j] * 2);
                }
                if (now > 1) {
                    dp[i][j][k][1][0] = max(dp[i][j][k][1][0],dp[i ^ 1][j][k][0][0] + a[now][j]);
                    dp[i][j][k][1][0] = max(dp[i][j][k][1][0],dp[i ^ 1][j][k][1][0] + a[now][j]);
                    dp[i][j][k][1][0] = max(dp[i][j][k][1][0],dp[i ^ 1][j][k][0][1] + a[now][j]);
                    dp[i][j][k][1][0] = max(dp[i][j][k][1][0],dp[i ^ 1][j][k][1][1] + a[now][j]);
                    if (k < kk) {
                        dp[i][j][k][1][1] = max(dp[i][j][k][1][1],dp[i ^ 1][j][k + 1][0][0] + a[now][j] * 2 + a[now-1][j]);
                        dp[i][j][k][1][1] = max(dp[i][j][k][1][1],dp[i ^ 1][j][k + 1][1][0] + a[now][j] * 2 + a[now-1][j]);
                        dp[i][j][k][1][1] = max(dp[i][j][k][1][1],dp[i ^ 1][j][k + 1][0][1] + a[now][j] * 2);
                        
                    }
                    dp[i][j][k][1][1] = max(dp[i][j][k][1][1],dp[i ^ 1][j][k][1][1] + a[now][j] * 2);
                    
                }
            }
        }
    int ans = 0;
    for (int k = 0;k <= kk;k ++ )
        for (int l = 0;l < 2;l ++ )
        for (int j = 0;j < 2;j ++ ) ans = max(ans,dp[i^1][m][k][l][j]);
    cout << ans;
    return 0;
}
```

## D 路径解析（加强版）

![](C:\Users\yangsonghan\my-blog\static\images\程设选拔考题解\T4.png)



$1\le p\le 10^5, 1\le L\le 10^6,2\le S\le 5\times 10^6, 2\le T\le 8\times 10^6$，其中 $L$  是单条路径长度最大值，$S$ 为所有串长度总和，$T$ 为输出长度总和。

简单模拟题。由于太久没有调这种题，被一些 RE 输出的乱码骗了，各种查不出错，最终爆 $0$ 离场。

没啥好说的，用 map 把每个目录映射到一个节点上，并标记每个目录的上级目录是谁。注意，由于在不同位置的目录名字可能相同，map 里需要用一个 $\text{pair<string,int>}$ 来存目录的名字以及其父亲（上层目录）的编号。先扫一遍 $c$ 把从根目录到 $c$ 的节点建出来。然后扫询问的时候把没出现过的节点建出来就好了。

然后就是简单处理一下不规范的情况，并且在最后递归地输出答案即可。

难点在于处理询问是一个空行的情况。用 $\text{scanf}$ 之类的不管用，要一个一个读字符并且判断第一个字符是不是换行。用 $\text{scanf}$ 读完字符串 $c$ 之后会剩一个空行，需要提前读掉。

代码中附有一组样例。

```c++
#include<bits/stdc++.h>
using namespace std;
const int N = 1e6;
typedef long long ll;
const int inf = 1e9;
int read() {
    int s = 0,f = 1;char ch = getchar();
    while (!isdigit(ch)) f = ch == '-' ? -1 : 1, ch = getchar();
    while (isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
    return s * f;
}
char c[N + 10],s[N + 10];
int n,m,fa[N + 10],tot;
map<pair<string,int>,int> mp;
string str[N + 10];
void print(int x) {
    if (x == 0) return;
    print(fa[x]);
    cout << '/' << str[x];
}
int main() {
    int T ; cin >> T;
    scanf("%s",c+1);
    m = strlen(c + 1);
    mp.clear();
    for (int i = 1;i <= m;i ++ ) {
        int j = i;
        string ss = "";
        while (j < m && c[j + 1] != '/') j ++,ss += c[j];
        //i+1~j
        tot ++;
        mp[{ss,tot - 1}] = tot;
        str[tot] = ss;
        fa[tot] = tot - 1;
        i = j;
    }
    int start = tot;
    getchar();
    while (T -- ) {
        char ch = getchar();
        if (ch == '\n') {
            print(start);
            puts("");
            continue;
        }
        s[1] = ch;n = 1;
        ch = getchar();
        while (ch != '\n' && ch != EOF) s[++n] = ch, ch = getchar();
        s[n + 1] = '\0'; 
        int now = start;
        if (s[1] == '/') now = 0;
        for (int i = 1;i <= n;i ++ ) {
            int j = i;
            while (j <= n && s[j] == '/') j ++ ;
            j--;
            //i~j : /
            int k = j;
            string ss = "";
            while (k < n && s[k + 1] != '/') k ++, ss += s[k];
            //j+1~k
            if (k == j) break;
            if (ss == ".") {
                i = k;
                continue;
            }
            else if (ss == "..") {
                now = fa[now];
                i = k;
                continue;
            }
            else {
                if (mp[{ss,now}] == 0) mp[{ss,now}] = ++tot, str[tot] = ss, fa[tot] = now;
                int x = mp[{ss,now}];
                now = x;
                i = k;
                continue;
            }
        }
        if (!now) {
            puts("/");
            continue;
        }
        print(now);
        puts("");
    }
    return 0;
}
/*
input
6
/a/b
../..././x
/..name//a..b/
../../../../
./c/../d
/./A/../a
x/../../y

output
/a/.../x
/..name/a..b
/
/a/b/d
/a
/a/y
*/
```

## E 记忆力训练（加强版）

![](C:\Users\yangsonghan\my-blog\static\images\程设选拔考题解\T5.png)

$1\le n\le 10^{12}, 1\le q\le 10^5,0\le d\le 2,1\le k_i\le n$。

注意到原整数最多只有 $12$ 位（$10^{12}$ 可以排除掉），并且新整数的质因子不会超过 $11$ （在 $d<2$ 时不会超过 $7$），于是考虑直接枚举所有新整数，并求出对应的原整数即可。枚举新整数的办法就是枚举其每个质因子的次数。

枚举完质因子，关键是怎么求新整数对应的原整数。这里 $d=2$ 要更复杂一些，先考虑 $d=0,1$。

原整数要求最小，因此首先尽量压缩位数。质因子 $7$ 没有操作空间，一个就是一位。在 $d=1$ 的时候，由于 $9+1=2\times 5$，因此可以把 $2$ 和 $5$ 合并来减少位数。对于质因子 $2,3$，比较容易想到的是 $3$ 个 $2$ 凑一起， $2$ 个 $3$ 凑一起使位数最小。那么位数相同的前提下优先使小的数更小，于是对于 $2,2,3$ 最优方案是拼成 $2,6$。这里注意此时对于 $4$ 个 $2$来说， $2,8$ 优于 $4,4$，而对于$2,3,3$来说， $2,9$ 优于 $3,6$，因此优先让$3$ 个 $2$ 凑一起， $2$ 个 $3$ 凑一起是正确的。这样一通操作下来，我们将原整数的每一位都确定下来，最后只需从小到大拼接即可。



可是，对于 $d=2$，在处理质因子 $2,3,5$ 的时候有变化。因为如果剩下单个 $2$，在原整数中会变成 $0$，而这不能作为首位，需要让第二大的移到首位。因此对于下列情况，我们需要做出改变：

- 当没有质因子 $3$，$2$ 和 $5$ 合并之后只剩下 $3n+1$ 个 $2$ 时，如果 $n\ge 1$，那么将 $2,8$ 变为 $4,4$ 将会更优（减 $2$ 之后变为 $0,6$ 和 $2,2$，由当前无 $3$ 可知前者首位必定大于 $2$，而后者在位数相同的情况下首位更小，于是更优，接下来的情况同理）。如果 $n=0$，如果之前有 $2\times 5$ 合并，那么拆掉一个 $2\times 5$，把 $2 \times 10$ 换成 $4,5$ 会更优。当然如果总共就一个 $2$，$3,5$ 都没有那这个 $0$ 就无法避免。

- 当有质因子 $3$，$2$ 和 $5$ 合并之后只剩下 $3n+1$ 个 $2$ 时，这个 $2$ 直接和 $3$ 合并即可。因为此时对于 $2,3,3$（即偶数个 $3$），$3,6$ 比 $2,9$ 更优（注意与之前的情况反过来了），对奇数个 $3$ 的情况，不和 $3$ 合并会导致位数更大。

一些边界条件：对新整数是 $1,2$ 要特殊处理。

然后注意我们要按新整数大小排序，于是要用 $\text{pair}$ 存原整数和新整数。

还有就是 $d=0$ 的时候 $0$ 也会被写到本子上，原整数是 $10$，特殊处理一下即可。

复杂度不知道。

```c++
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e6;
const int inf = 1e9;
const ll INF = 1e12;
ll read() {
    ll s = 0,f = 1;char ch = getchar();
    while (!isdigit(ch)) f = ch == '-'? -1 : 1, ch = getchar();
    while(isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
    return s * f;
}
ll n;
int d,q,cnt;
int pr[5] = {2,3,5,7,11};
int tmp[5];
int num[100];
pair<ll,ll> S[N + 10];
void get() {
    int c2 = tmp[0],c3 = tmp[1],c5 = tmp[2],c7 = tmp[3],c11 = tmp[4];
    int tot = 0,c2_1 = 0;
    while (c2 && c5 && d >= 1) c2 --, c5 --, num[++tot] = 10 - d;
    c2_1=c2;
    for (int i = 1;i <= c2 / 3;i ++ ) num[++tot] = 8 - d;
    c2 %= 3;
    if (d == 2) {
        
        if (!c3 && !c5 && c2_1 >= 4 && c2 == 1) {//no 3,  2 2 2 2
            tot --;
            num[++tot] = 4 - d;
            num[++tot] = 4 - d;
            c2 = 0;
        }
        else if (tmp[2] && c2 == 1 && !c3) {//no 3, 2 2 5
            num[1] = 4 - d;
            c5 ++, c2 = 0;
        }
        else if (c2 == 1 && c3) {//3, 2 2 2 2
            c2 --, c3 --, num[++tot] = 6 - d;
        }
    }
    if (d < 2) {
        for (int i = 1;i <= c3 / 2;i ++ ) num[++tot] = 9 - d;
        c3 %= 2;
        if (c2 && c3) c2 --, c3 --, num[++tot] = 6 - d;
    }
    else {
        if (c2 == 1 && c3) c2 --, c3 --, num[++tot] = 6 - d;
        for (int i = 1;i <= c3 / 2;i ++ ) num[++tot] = 9 - d;
        c3 %= 2;
    }
    if (c2 == 1) num[++tot] = 2 - d;
    else if (c2 == 2) num[++tot] = 4 - d;
    if (c3 == 1) num[++tot] = 3 - d;
    for (int i = 1;i <= c5;i ++ ) num[++tot] = 5 - d;
    for (int i = 1;i <= c7;i ++ ) num[++tot] = 7 - d;
    if (d == 2) for (int i = 1;i <= c11;i ++ ) num[++tot] = 11 - d;
    if (!tot) {
        if (!d) S[++cnt] = {1,1};
        return;
    }
    if (tot > 12) return;
    sort (num + 1,num + tot + 1);
    if (tot == 1 && num[1] == 0) return;
    if (num[1] == 0) swap(num[1],num[2]);
    ll ss = 0;
    for (int i = 1;i <= tot;i ++ ) ss = ss * 10 + num[i];
    if (ss <= n) {
        ll ori = 1;
        for (int i = 0;i < 5;i ++ )
            for (int j = 0;j < tmp[i];j ++ ) ori *= pr[i];
        S[++cnt] = {ori,ss};
    }
}
int main() {
    n = read(), d = read(), q = read();
    for (int i = 0;i <= 36;i ++ ) {
        tmp[0] = i;
        for (int j = 0;j <= 24;j ++ ) {
            tmp[1] = j;
            for (int k = 0;k <= 12;k ++ ) {
                tmp[2] = k;
                for (int l = 0;l <= 12;l ++ ) {
                    tmp[3] = l;
                    if (d < 2) {
                        get();
                    }
                    else {
                        for (int m = 0;m <= 12;m ++ ) {
                            tmp[4] = m;
                            get();
                        }
                    }
                    
                }
            } 
                
        }
    }
    if (d == 0 && n >= 10) S[++cnt] = {0,10};
    sort(S + 1,S + cnt + 1);
    while (q -- ) {
        ll k = read();
        if (k > cnt) cout << -1 << ' ';
        else printf("%lld ",S[k].second);
    }
    return 0;
}
```
